DROP TABLE IF EXISTS  people;

CREATE TABLE people (
	personID INTEGER PRIMARY KEY,
	firstName VARCHAR(30) NOT NULL,
	lastName VARCHAR(30),
	birthdate DATE NOT NULL,
	planner INTEGER,
	email VARCHAR(30),
	FOREIGN KEY(planner) REFERENCES people(personID)
);
